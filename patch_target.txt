                <div className="bg-blue-50/30 p-4 rounded-xl border border-blue-100 grid grid-cols-1 md:grid-cols-3 gap-4 relative">
                  <div className="col-span-1 md:col-span-3 mb-2 border-b border-blue-100 pb-2 flex justify-between items-center">
                    <h3 className="text-sm font-bold text-[#0B1120]">2. Configuración de Costos y Empaque</h3>
                  </div>

                  {/* EMPAQUE PRINCIPAL */}
                  <div className="bg-white p-3 rounded-lg border border-gray-200 shadow-sm col-span-1 md:col-span-3 grid grid-cols-1 sm:grid-cols-3 gap-4">
                    <div>
                      <label className="block text-[10px] font-bold text-gray-500 mb-1 uppercase tracking-wide">Presentación Principal</label>
                      <select 
                        value={newProduct.empaqueBase || 'UNIDAD'} 
                        onChange={e => {
                          setNewProduct({
                            ...newProduct, 
                            empaqueBase: e.target.value,
                            ventaPorDefecto: 'UND'
                          });
                        }} 
                        className="w-full p-2 border border-gray-200 rounded-lg text-xs font-bold bg-gray-50 focus:bg-white focus:outline-none focus:border-[#2563EB]"
                      >
                        {(companyConfig?.packagingTypes || ['CAJA', 'PIEZA', 'JUEGO', 'UNIDAD', 'PAQUETE']).map(pkg => (
                          <option key={pkg} value={pkg}>{pkg}</option>
                        ))}
                      </select>
                    </div>

                    <div className="flex items-center mt-4">
                      <label className="flex items-center cursor-pointer group">
                        <input 
                          type="checkbox" 
                          checked={newProduct.unidadesPorEmpaque > 1}
                          onChange={(e) => {
                            if (e.target.checked) {
                              setNewProduct({ ...newProduct, unidadesPorEmpaque: 2 });
                            } else {
                              setNewProduct({ ...newProduct, unidadesPorEmpaque: 1, precioUnidadDetal: 0 });
                            }
                          }}
                          className="w-4 h-4 text-[#2563EB] bg-gray-100 border-gray-300 rounded focus:ring-[#2563EB] cursor-pointer"
                        />
                        <span className="ml-2 text-xs font-bold text-gray-700 uppercase tracking-wide group-hover:text-[#2563EB] transition-colors">
                          Lleva unidad secundaria
                        </span>
                      </label>
                    </div>

                    {newProduct.unidadesPorEmpaque > 1 && (
                      <div>
                        <label className="block text-[10px] font-bold text-gray-500 mb-1 uppercase tracking-wide">¿Cuántas UND trae la {newProduct.empaqueBase || 'UNIDAD'}?</label>
                        <input 
                          type="number" min="2" 
                          value={newProduct.unidadesPorEmpaque || ''} 
                          onChange={e => {
                            const val = parseInt(e.target.value) || 1;
                            const currentCostoUnidad = newProduct.costoDolar || 0;
                            setNewProduct({
                              ...newProduct, 
                              unidadesPorEmpaque: val,
                              costoEmpaque: Number((currentCostoUnidad * val).toFixed(2))
                            });
                          }} 
                          className="w-full p-2 border border-gray-200 rounded-lg text-xs font-bold text-blue-600 bg-gray-50 focus:bg-white focus:outline-none focus:border-[#2563EB]" 
                        />
                      </div>
                    )}
                    
                    {!isEditMode && (
                    <div>
                      <label className="block text-[10px] font-bold text-gray-500 mb-1 uppercase tracking-wide">
                        Stock Inicial ({newProduct.unidadesPorEmpaque > 1 ? `en ${newProduct.empaqueBase || 'UNIDAD'}s` : 'en UND'})
                      </label>
                      <input 
                        type="number" min="0" 
                        value={newProduct.stockDisp !== undefined && newProduct.unidadesPorEmpaque ? (newProduct.stockDisp / newProduct.unidadesPorEmpaque) : 0}
                        onChange={e => {
                          const val = Number(e.target.value) || 0;
                          const factor = newProduct.unidadesPorEmpaque || 1;
                          setNewProduct({...newProduct, stockDisp: (val * factor)});
                        }} 
                        className="w-full p-2 border border-gray-200 rounded-lg text-xs font-bold text-gray-800 bg-gray-50 focus:bg-white focus:outline-none" 
                      />
                      {newProduct.unidadesPorEmpaque > 1 && newProduct.stockDisp > 0 && (
                        <div className="text-[9px] text-gray-400 mt-1 font-bold">
                          = {newProduct.stockDisp} unidades sueltas
                        </div>
                      )}
                    </div>
                    )}
                  </div>

                  {/* COSTOS BIDIRECCIONALES */}
                  <div className={`col-span-1 md:col-span-3 grid grid-cols-1 ${newProduct.unidadesPorEmpaque > 1 ? 'sm:grid-cols-2' : ''} gap-4`}>
                    <div>
                      <label className="block text-[10px] font-bold text-gray-500 mb-1 uppercase tracking-wide">
                        Costo por {newProduct.empaqueBase || 'UNIDAD'} (USD)
                      </label>
                      <input 
                        type="number" step="0.01" min="0" required
                        value={newProduct.costoEmpaque || ''} 
                        onChange={e => {
                          const val = Number(e.target.value);
                          const divisor = newProduct.unidadesPorEmpaque || 1;
                          setNewProduct({
                            ...newProduct, 
                            costoEmpaque: val,
                            costoDolar: Number((val / divisor).toFixed(2)) // Redondeado a 2 decimales
                          });
                        }} 
                        className="w-full p-2 border border-gray-200 rounded-lg text-xs font-bold text-gray-800 bg-white focus:outline-none focus:ring-2 focus:ring-[#2563EB]/20" 
                      />
                    </div>

                    {newProduct.unidadesPorEmpaque > 1 && (
                    <div>
                      <label className="block text-[10px] font-bold text-gray-500 mb-1 uppercase tracking-wide">
                        Costo Unitario (USD)
                      </label>
                      <input 
                        type="number" step="0.001" min="0"
                        value={newProduct.costoDolar || ''} 
                        onChange={e => {
                          const val = Number(e.target.value);
                          const factor = newProduct.unidadesPorEmpaque || 1;
                          setNewProduct({
                            ...newProduct, 
                            costoDolar: val,
                            costoEmpaque: Number((val * factor).toFixed(2))
                          });
                        }} 
                        className="w-full p-2 border border-gray-200 rounded-lg text-xs font-bold text-gray-500 bg-gray-50 focus:outline-none focus:ring-2 focus:ring-[#2563EB]/20" 
                      />
                    </div>
                    )}
                  </div>
                </div>

                {/* SECCION 3: PRECIOS Y RENTABILIDAD */}
                <div className="bg-emerald-50/30 p-4 rounded-xl border border-emerald-100 grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div className="col-span-1 md:col-span-2 mb-2 border-b border-emerald-100 pb-2">
                    <h3 className="text-sm font-bold text-[#0B1120]">3. Precios de Venta</h3>
                  </div>

                  <div>
                    <div className="flex justify-between mb-1">
                      <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wide">
                        Precio Mayorista - {newProduct.empaqueBase || 'UNIDAD'}
                      </label>
                      <span className="text-[10px] font-bold text-emerald-600">
                        Mg: {newProduct.costoEmpaque && newProduct.precioMayor ? (((newProduct.precioMayor - newProduct.costoEmpaque) / newProduct.costoEmpaque) * 100).toFixed(0) : '0'}%
                      </span>
                    </div>
                    <input 
                      required type="number" step="0.01" min="0" 
                      value={newProduct.precioMayor || ''} 
                      onChange={e => setNewProduct({...newProduct, precioMayor: Number(e.target.value)})} 
                      className="w-full p-2 border border-emerald-200 rounded-lg text-xs font-black text-emerald-700 bg-white focus:outline-none focus:ring-2 focus:ring-emerald-500/20" 
                    />
                  </div>

                  <div>
                    <div className="flex justify-between mb-1">
                      <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wide">
                        Precio Minorista - {newProduct.empaqueBase || 'UNIDAD'}
                      </label>
                      <span className="text-[10px] font-bold text-emerald-600">
                        Mg: {newProduct.costoEmpaque && newProduct.precioMinor ? (((newProduct.precioMinor - newProduct.costoEmpaque) / newProduct.costoEmpaque) * 100).toFixed(0) : '0'}%
                      </span>
                    </div>
                    <input 
                      required type="number" step="0.01" min="0" 
                      value={newProduct.precioMinor || ''} 
                      onChange={e => setNewProduct({...newProduct, precioMinor: Number(e.target.value)})} 
                      className="w-full p-2 border border-emerald-200 rounded-lg text-xs font-black text-emerald-700 bg-white focus:outline-none focus:ring-2 focus:ring-emerald-500/20" 
                    />
                  </div>

                  {newProduct.unidadesPorEmpaque > 1 && (
                    <div className="col-span-1 md:col-span-2 mt-2 pt-4 border-t border-emerald-100/50">
                      <div className="flex justify-between mb-1">
                        <label className="block text-[10px] font-bold text-gray-500 uppercase tracking-wide">
                          Precio Unidad Secundaria
                        </label>
                        <span className="text-[10px] font-bold text-emerald-600">
