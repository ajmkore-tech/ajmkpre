   461	                      <div className="flex flex-col gap-1.5">
   462	                        <div className="text-[10px] text-gray-500 font-medium">{new Date(sale.fecha).toLocaleDateString()} &middot; {sale.items.length} items</div>
   463	                        <div className="flex items-center gap-1 text-[9px] font-bold text-amber-600 bg-amber-50 w-fit px-1.5 py-0.5 rounded border border-amber-100">
   464	                          <Clock className="w-3 h-3" />
   465	                          <span>{getDaysSince(sale.fechaEstado || sale.fecha)} en este estado</span>
   466	                        </div>
   467	                      </div>
   468	                      <div className="text-sm font-black text-[#10B981]">${sale.total.toFixed(2)}</div>
   469	                    </div>
   470	                  </div>
   471	                ))
   472	              )}
   473	            </div>
   474	          </div>
   475	        </div>
   476	
   477	        {/* Detail View */}
   478	        {selectedSale ? (
   479	          <div className="flex-1 flex flex-col bg-[#F8FAFC] overflow-hidden relative">
   480	            <div className="flex-1 overflow-y-auto custom-scrollbar p-4 lg:p-6">
   481	              <button onClick={() => setSelectedSale(null)} className="lg:hidden mb-4 text-sm font-bold text-gray-500 hover:text-[#2563EB] flex items-center gap-1 transition-colors"><ArrowLeft className="w-4 h-4" /> Volver a Pedidos</button>
   482	              
   483	              <div className="flex flex-col xl:flex-row xl:justify-between xl:items-start gap-4 mb-4">
   484	                <div>
   485	                  <div className="flex items-center gap-3 mb-2 flex-wrap">
   486	                    <h2 className="text-2xl font-black text-[#0B1120] font-mono">{selectedSale.id}</h2>
   487	                    {selectedSale.numeroDocumento && (
   488	                      <span className="text-sm font-bold text-gray-500 bg-white px-2 py-0.5 rounded border border-gray-200">
   489	                        {selectedSale.tipoDocumento === 'Factura' ? 'Factura' : 'Nota'} #{selectedSale.numeroDocumento}
   490	                      </span>
   491	                    )}
   492	                    <div className={`flex items-center gap-1 px-2.5 py-1 rounded-full text-[10px] font-bold border uppercase tracking-wider ${getStatusColor(selectedSale.estado)}`}>
   493	                      {getStatusIcon(selectedSale.estado)} {selectedSale.estado}
   494	                    </div>
   495	                  </div>
   496	                  <div className="text-sm font-bold text-gray-600">{new Date(selectedSale.fecha).toLocaleString()}</div>
   497	                </div>
   498	                
   499	                <div className="flex flex-wrap items-center gap-2">
   500	                  {canPrintDocument(selectedSale) && (
   501	                    <button 
   502	                      onClick={() => generateDocument(selectedSale, selectedSale, setSelectedSale)}
   503	                      className="bg-indigo-50 text-indigo-700 px-3 py-2 rounded-lg text-xs font-bold flex items-center gap-2 hover:bg-indigo-100 transition-all border border-indigo-100"
   504	                      title="Imprimir Documento"
   505	                    >
   506	                      <FileText className="w-3.5 h-3.5" /> {selectedSale.numeroDocumento ? 'Ver Documento' : 'Generar PDF'}
   507	                    </button>
   508	                  )}
   509	                  {selectedSale.estado !== 'Cerrado' && (
   510	                    <button 
   511	                      onClick={() => handleDeleteSale(selectedSale.id)}
   512	                      className="bg-red-50 text-red-600 px-3 py-2 rounded-lg text-xs font-bold flex items-center gap-2 hover:bg-red-100 transition-all border border-red-100"
   513	                      title="Eliminar pedido y reversar inventario"
   514	                    >
   515	                      <Trash2 className="w-3.5 h-3.5" /> Eliminar
   516	                    </button>
   517	                  )}
   518	                  {getNextStatusAction(selectedSale) && (
   519	                    <button 
   520	                      onClick={() => advanceStatus(selectedSale.id)}
   521	                      className="bg-[#0B1120] text-white px-4 py-2 rounded-lg text-xs font-bold flex items-center gap-2 hover:bg-gray-800 transition-all shadow-sm"
   522	                    >
   523	                      {getNextStatusAction(selectedSale)} <ArrowRight className="w-3.5 h-3.5" />
   524	                    </button>
   525	                  )}
   526	                </div>
   527	              </div>
   528	              
   529	              {selectedSale.anulado && <div className="bg-red-600 text-white font-black text-center py-2 px-4 rounded-xl mb-4 uppercase tracking-wider text-sm">ANULADO POR {selectedSale.anuladoPor?.toUpperCase()}</div>}
   530	
   531	              {/* Unified Document Block */}
   532	              <div className="bg-white border border-gray-200 rounded-2xl shadow-sm overflow-hidden flex flex-col mb-6">
   533	                 
   534	                 {/* Top Section: Client condensed */}
   535	                 <div className="p-4 border-b border-gray-100 flex justify-between items-start sm:items-center bg-gray-50/50 flex-col sm:flex-row gap-3">
   536	                    <div>
   537	                      <div className="flex items-center gap-2">
   538	                        <h3 className="text-sm font-black text-[#0B1120] uppercase tracking-wider">{selectedSale.clienteNombre}</h3>
   539	                        {selectedSale.estado !== 'Cerrado' && (
   540	                           <button onClick={() => handleEditSale(selectedSale)} className="p-1 hover:bg-gray-200 text-gray-500 rounded transition-colors" title="Editar Pedido">
   541	                              <FileEdit className="w-3.5 h-3.5" />
   542	                           </button>
   543	                        )}
   544	                      </div>
   545	                      <div className="text-[10px] text-gray-500 font-bold mt-0.5 uppercase tracking-wider">
   546	                         RIF: {clients.find(c => c.id === selectedSale.clienteId)?.rif || selectedSale.clienteId} • {clients.find(c => c.id === selectedSale.clienteId)?.telefono || 'S/N'}
   547	                      </div>
   548	                    </div>
   549	                    <div className="flex flex-row sm:flex-col items-start sm:items-end gap-1">
   550	                       <span className="bg-blue-50 text-blue-700 border border-blue-200 px-2 py-0.5 rounded text-[9px] font-bold uppercase">{selectedSale.tipoPrecio}</span>
   551	                       <span className="bg-emerald-50 text-emerald-700 border border-emerald-200 px-2 py-0.5 rounded text-[9px] font-bold uppercase">{selectedSale.condicionPago}</span>
   552	                    </div>
   553	                 </div>
   554	
   555	                 {/* Articles Table */}
   556	                 <div className="overflow-x-auto">
   557	                    <table className="w-full text-left text-xs">
   558	                      <thead className="bg-[#F8FAFC] border-b border-gray-200 text-gray-500 uppercase font-bold text-[9px] tracking-wider">
   559	                        <tr>
   560	                          <th className="px-3 py-2">Código</th>
   561	                          <th className="px-3 py-2">Descripción</th>
   562	                          <th className="px-3 py-2 text-right">Cant.</th>
   563	                          <th className="px-3 py-2 text-right">P. Unit</th>
   564	                          <th className="px-3 py-2 text-right text-[#0B1120]">Total</th>
   565	                        </tr>
   566	                      </thead>
   567	                      <tbody className="divide-y divide-gray-100">
   568	                        {selectedSale.items.map((item, idx) => (
   569	                          <tr key={idx} className="hover:bg-gray-50 transition-colors">
   570	                            <td className="px-3 py-2 font-mono font-bold text-[#2563EB]">{item.codigo}</td>
   571	                            <td className="px-3 py-2 min-w-[200px]">
   572	                              <div className="font-medium text-gray-800">{item.detalle}</div>
   573	                              <div className="text-[9px] text-gray-400 mt-0.5">{item.marcaRepuesto} &middot; {item.marcaVehiculo}</div>
   574	                            </td>
   575	                            <td className="px-3 py-2 text-right font-bold text-gray-600">{item.cantidad}</td>
   576	                            <td className="px-3 py-2 text-right text-gray-600">${item.precio.toFixed(2)}</td>
   577	                            <td className="px-3 py-2 text-right font-black text-[#10B981]">${item.total.toFixed(2)}</td>
   578	                          </tr>
   579	                        ))}
   580	                      </tbody>
   581	                    </table>
   582	                 </div>
   583	
   584	                 {/* Totals Section */}
   585	                 <div className="p-4 bg-gray-50/50 border-t border-gray-200 flex flex-col sm:flex-row sm:items-center justify-end gap-6 shrink-0">
   586	                    {selectedSale.tipoDocumento === 'Factura' && (
   587	                      <div className="flex items-center justify-end gap-6">
   588	                        <div className="flex flex-col text-right">
   589	                          <span className="text-[9px] font-bold text-gray-500 uppercase">Subtotal</span>
   590	                          <span className="text-sm font-bold text-gray-700">${(selectedSale.total / (1 + (companyConfig.ivaVenta ?? 16) / 100)).toFixed(2)}</span>
   591	                        </div>
   592	                        <div className="flex flex-col text-right">
   593	                          <span className="text-[9px] font-bold text-gray-500 uppercase">IVA ({companyConfig.ivaVenta ?? 16}%)</span>
   594	                          <span className="text-sm font-bold text-gray-700">${(selectedSale.total - (selectedSale.total / (1 + (companyConfig.ivaVenta ?? 16) / 100))).toFixed(2)}</span>
   595	                        </div>
   596	                      </div>
   597	                    )}
   598	                    <div className="flex flex-col text-right sm:pl-6 sm:border-l sm:border-gray-200 mt-2 sm:mt-0">
   599	                      <span className="text-[9px] font-bold text-gray-500 uppercase">Total {selectedSale.tipoDocumento === 'Factura' ? '(con IVA)' : ''}</span>
   600	                      <div className="flex items-baseline gap-2 justify-end">
   601	                        <span className="text-2xl font-black text-[#10B981] leading-none">${selectedSale.total.toFixed(2)}</span>
   602	                        <span className="text-[10px] font-bold text-gray-400">Bs. {(selectedSale.total * (selectedSale.tasaAplicada || 1)).toLocaleString('de-DE', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</span>
   603	                      </div>
   604	                    </div>
   605	                 </div>
   606	              </div>
   607	              
   608	              <div className="bg-white border border-gray-200 rounded-xl p-4 shadow-sm">
   609	                <h4 className="text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-2">Auditoría</h4>
   610	                <div className="space-y-1">
   611	                  {selectedSale.auditLog?.map((log, idx) => (
   612	                    <div key={idx} className="flex gap-4 text-[10px]">
   613	                      <span className="text-gray-400 min-w-[120px] font-mono">{new Date(log.date).toLocaleString('es-VE')}</span>
   614	                      <span className="text-gray-700 font-medium">{log.action}</span>
   615	                    </div>
   616	                  )) || <div className="text-[10px] text-gray-400 italic">No hay registros de auditoría</div>}
   617	                </div>
   618	              </div>
   619	                                      </div>
   620	          </div>
   621	        ) : (
   622	          <div className="hidden lg:flex flex-1 items-center justify-center bg-gray-50">
   623	            <div className="text-center">
   624	              <div className="bg-white w-16 h-16 rounded-full flex items-center justify-center shadow-sm border border-gray-100 mx-auto mb-4">
   625	                <CheckCircle2 className="w-8 h-8 text-gray-300" />
   626	              </div>
   627	              <h3 className="text-lg font-black text-gray-400">Selecciona un pedido</h3>
   628	              <p className="text-xs text-gray-400 mt-1">El detalle de la orden aparecerá aquí.</p>
   629	            </div>
   630	          </div>
   631	        )}
   632	      </div>
   633	
   634	      {/* Nuevo Pedido Modal */}
   635	      {isModalOpen && <NewOrderModal state={modalState} />}
   636	
   637	
   638	      {/* Closing Modal */}
   639	      {isClosingModalOpen && closingSale && <CloseSaleModal state={modalState} />}
   640	
   641	
   642	    </div>
   643	  );
   644	}
